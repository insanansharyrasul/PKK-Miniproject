\begin{frame}{Latar Belakang \& Formulasi Masalah}
  \framesubtitle{Optimasi Rute Logistik \textit{Traveling Salesperson Problem}}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Latar Belakang}
    \begin{itemize}
      \item Optimasi rute krusial untuk sektor logistik dan distribusi guna meminimalkan biaya operasional.
      \item Menggunakan dataset riil 27 kabupaten/kota di Jawa Barat (Mencakup Nama, Latitude (X), Longitude (Y)).
      \item Ruang pencarian berskala eksponensial $\approx 27!$ ($10^{28}$) kemungkinan rute, sehingga tergolong masalah \textbf{NP-Hard}.
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Formulasi Masalah}
    \begin{itemize}
      \item Diberikan sekumpulan lokasi $V=\{v_1,v_2,\dots,v_n\}$.
      \item \textbf{Tujuan TSP:} Menemukan rute siklik yang mengunjungi setiap kota tepat satu kali dengan akumulasi jarak tempuh seminimal mungkin.
      \item \textbf{Fungsi Objektif:}
      \[
        \min \sum_{i=1}^{n-1} d(\pi_i,\pi_{i+1})+d(\pi_n,\pi_1)
      \]
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Sebaran dan Perhitungan Jarak}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Kalkulasi Jarak (Formula Haversine)}
    \begin{itemize}
      \item Digunakan untuk menghitung jarak geodesik antar dua titik koordinat:
      \vspace{-0.1cm}
      {\footnotesize
      \[ a = \sin^2\!\left(\frac{\Delta\phi}{2}\right) + \cos(\phi_1)\cos(\phi_2)\sin^2\!\left(\frac{\Delta\lambda}{2}\right) \]
      \[ c = 2 \cdot \text{atan2}\left(\sqrt{a}, \sqrt{1-a}\right), \quad d = R \cdot c \]
      }
      \vspace{-0.2cm}
      \item \textbf{Mengapa Haversine?} Jauh lebih presisi dibandingkan metrik \textit{Euclidean} datar karena mengakomodasi kelengkungan bola bumi.
    \end{itemize}
    
    \column{0.48\textwidth}
    \centering
    \includegraphics[width=0.85\linewidth,height=0.55\textheight,keepaspectratio]{04_matriks_jarak_geografis.png}
    
    \vspace{0.2cm}
    \justifying
    {\scriptsize \textbf{Interpretasi Heatmap:}
    \begin{itemize}
      \item Matriks jarak \textit{pairwise} membuktikan adanya kluster wilayah dominan (mis. kluster Bandung Raya yang ditandai warna kuning/gelap pekat).
      \item Mengindikasikan bahwa rute TSP yang optimal akan memprioritaskan penyelesaian kunjungan di dalam satu \textit{cluster} sebelum berpindah wilayah.
    \end{itemize}
    }
  \end{columns}
\end{frame}

\begin{frame}{Baseline: \textit{Greedy Nearest Neighbor}}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Pendekatan Pembanding (Heuristik Lokal)}
    \begin{enumerate}
      \item Memulai dari kota awal.
      \item Selalu memilih kota terdekat yang belum dikunjungi.
      \item Mengulangi hingga seluruh kota dikunjungi, lalu kembali ke titik awal.
    \end{enumerate}
    
    \vspace{0.2cm}
    \textbf{Kelebihan:} Cepat (Kompleksitas $\mathcal{O}(n^2)$) dan sederhana pelaksanaannya.
    
    \vspace{0.2cm}
    \textbf{Kekurangan:}
    \begin{itemize}
      \item Pemilihan solusi bersifat proksimal-lokal.
      \item Sangat rentan terjebak \textit{local optimum}.
      \item Mengabaikan konsekuensi jangka panjang (menyisakan kota terjauh di akhir sehingga melahirkan lompatan jarak jauh).
    \end{itemize}
    
    \column{0.45\textwidth}
    \vspace{0.5cm}
    \begin{beamercolorbox}[wd=\linewidth,sep=12pt,rounded=true]{block body}
      \centering
      {\large\bfseries Hasil Baseline}\\[8pt]
      Total jarak: \textbf{\color{Sky} 891,94 km}
    \end{beamercolorbox}
  \end{columns}
\end{frame}

\begin{frame}{Mengapa Memetic PSO?}
  \framesubtitle{Relevansi Hibridisasi Metaheuristik dalam Masalah TSP}
  \begin{block}{Relevansi Hibridisasi \textit{Memetic}}
    \begin{itemize}
      \item \textbf{Kelemahan Tunggal:} Algoritma PSO murni lambat konvergen, sedangkan heuristik \textit{Greedy} murni rentan cacat di sisa akhir rutenya.
      \item \textbf{Sinergi \textit{Memetic}:} \textit{Greedy} menyemai rute awal yang cerdas, lalu PSO memperbaiki dan mengeksplorasinya secara global lintas partikel.
    \end{itemize}
  \end{block}
  
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{\textit{Particle Swarm Optimization} (PSO)} adalah metaheuristik pencarian global berbasis populasi. Tangguh dalam mendiversifikasi ruang pencarian dan melarikan diri dari jebakan \textit{local optimum}.
    \item \textbf{Heuristik Lokal (\textit{Greedy} \& \textit{2-Opt})} sangat unggul dalam eksploitasi mikro (menyediakan solusi awalan logis dan memoles cacat rute persilangan).
  \end{itemize}
\end{frame}

% ─────────────────────────────────────────────
%  SLIDE 7 · Adaptasi PSO
% ─────────────────────────────────────────────
\begin{frame}{Adaptasi PSO untuk TSP}
  \framesubtitle{\textit{Swap Operator} (Aljabar Diskrit)}
  Karena TSP berupa permutasi urutan kota (diskrit), PSO kontinu dimodifikasi menggunakan aljabar \textit{Swap Operator} [2]:
  
  \vspace{0.1cm}
  \begin{enumerate}\small
    \item \textbf{Posisi ($X$):} Representasi urutan kunjungan kota (Contoh: \texttt{[A, B, C, D]}).
    \item \textbf{\textit{Swap Operation} $SO(i, j)$:} pertukaran elemen pada indeks $i$ dan $j$.
    \item \textbf{\textit{Velocity} ($V$):} barisan berurutan dari operasi \textit{swap}: $V = [SO(i_1, j_1), SO(i_2, j_2), \dots]$
    \item \textbf{Pengurangan posisi} ($X_2 - X_1$): barisan \textit{swap} yang mengubah $X_1$ menjadi $X_2$.
    \item \textbf{Perkalian skalar} ($p \cdot V$): mempertahankan setiap \textit{swap} dengan probabilitas $p \in [0, 1]$.
  \end{enumerate}

  \begin{block}{Pembaruan Kecepatan \& Posisi}
    \centering
    $V_{t+1} = wV_t + c_1 r_1 (P_{\text{best}} - X_t) + c_2 r_2 (G_{\text{best}} - X_t)$ \\
    {\scriptsize di mana $w$ adalah bobot inersia, $c_1, c_2$ adalah faktor kognitif dan sosial, $r_1, r_2$ adalah bilangan acak $[0, 1]$.} \\[0.1cm]
    $X_{t+1} = X_t + V_{t+1}$
  \end{block}
\end{frame}

% ─────────────────────────────────────────────
\begin{frame}{Rancangan Memetic PSO}
  \framesubtitle{5 Strategi Optimasi Hibrida}
  Untuk meningkatkan kualitas solusi dan mencegah konvergensi prematur, PSO dipadukan dengan lima teknik optimasi:
  \vspace{0.2cm}
  \begin{table}[]
    \centering
    \scriptsize
    \begin{tabular}{@{}p{2.5cm}p{4cm}p{6.5cm}@{}}
      \toprule
      \textbf{Nama Strategi} & \textbf{Tujuan} & \textbf{Mekanisme} \\
      \midrule
      \textbf{Heuristic Seeding} & Mempercepat konvergensi di awal iterasi. & Partikel ke-0 disemai dengan rute \textit{Greedy}, sementara sisanya disebar secara acak.\\
      \addlinespace
      \textbf{Velocity Cap} & Menjaga stabilitas gerak partikel. & Membatasi panjang instruksi \textit{swap} maksimal ($\lfloor N/2 \rfloor$) per iterasi.\\
      \addlinespace
      \textbf{Periodic 2-opt} & Mengeliminasi \textit{crossing edge} (jalur saling menyilang). & Pencarian lokal membalikkan segmen rute, diterapkan pada 25\% partikel \textit{elite} setiap 5 iterasi.\\
      \addlinespace
      \textbf{Adaptive Mutation} & Transisi halus proses pencarian. & Laju probabilitas mutasi diturunkan secara linier (dari fase eksplorasi ke fase eksploitasi).\\
      \addlinespace
      \textbf{Partial Restart} & Menghindari stagnasi total di jebakan \textit{local optimum}. & Reinisialisasi 25\% partikel dengan kinerja terburuk apabila diversitas kawanan (CoV) turun di bawah $1\%$.\\
      \bottomrule
    \end{tabular}
  \end{table}
\end{frame}
